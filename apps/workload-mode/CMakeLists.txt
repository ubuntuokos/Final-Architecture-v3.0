cmake_minimum_required(VERSION 3.24)
project(WorkloadMode VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core DBus Widgets)
qt_standard_project_setup(REQUIRES 6.4)
add_executable(workmoded src/main.cpp src/ModeManager.cpp src/ModeManager.h)
add_executable(workmodectl src/workmodectl.cpp)
add_executable(workmoderun src/workmoderun.cpp)
add_executable(fa3-workload-mode-bridge src/fa3_workload_mode_bridge.cpp)
add_executable(workmode-indicator src/indicator.cpp)
foreach(t workmoded workmodectl workmoderun fa3-workload-mode-bridge)
  target_link_libraries(${t} PRIVATE Qt6::Core Qt6::DBus)
endforeach()
target_link_libraries(workmode-indicator PRIVATE Qt6::Widgets Qt6::DBus)
install(TARGETS workmoded fa3-workload-mode-bridge workmode-indicator RUNTIME DESTINATION libexec)
install(TARGETS workmodectl workmoderun RUNTIME DESTINATION bin)
